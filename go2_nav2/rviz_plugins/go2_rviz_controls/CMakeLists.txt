cmake_minimum_required(VERSION 3.16)
project(go2_rviz_controls)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
add_library(${PROJECT_NAME} SHARED src/speed_panel.cpp include/go2_rviz_controls/speed_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} rclcpp rcl_interfaces rviz_common pluginlib)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_speed_panel test/test_speed_panel.cpp)
  target_link_libraries(test_speed_panel ${PROJECT_NAME} Qt5::Test)
  ament_target_dependencies(test_speed_panel rclcpp rcl_interfaces rviz_common)
  add_test(NAME speed_panel_integration COMMAND test_speed_panel)
  set_tests_properties(speed_panel_integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=89;ROS_LOCALHOST_ONLY=1;CYCLONEDDS_URI=")
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_package()
